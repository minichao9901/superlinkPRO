cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico2 CACHE STRING "Pico SDK board for RP2350 Arm image")
set(PICO_NO_PICOTOOL 1)
set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}" CACHE PATH "Pico SDK 2.2.0 checkout")
if(NOT EXISTS "${PICO_SDK_PATH}/pico_sdk_init.cmake")
    message(FATAL_ERROR "Set PICO_SDK_PATH to an initialized Pico SDK checkout")
endif()
include("${PICO_SDK_PATH}/pico_sdk_init.cmake")
project(rp2350_workbench C CXX ASM)
pico_sdk_init()

set(RTT "${CMAKE_CURRENT_LIST_DIR}/../../web/tools/target-firmware/stm32f103_rtt_speed/segger_rtt")
add_executable(rp2350_workbench main.c "${RTT}/SEGGER_RTT.c")
target_include_directories(rp2350_workbench PRIVATE "${RTT}")
target_compile_options(rp2350_workbench PRIVATE -Og -g3 -gdwarf-4)
target_compile_definitions(rp2350_workbench PRIVATE BUFFER_SIZE_UP=32768)
target_link_libraries(rp2350_workbench PRIVATE pico_stdlib)
pico_enable_stdio_uart(rp2350_workbench 0)
pico_enable_stdio_usb(rp2350_workbench 0)
